% 编译：xelatex -interaction=nonstopmode research_flow.tex

\documentclass[border=6pt]{standalone}
\input{../../../references/assets/tikz/preamble.tex}
\input{../../../references/assets/tikz/flowchart-styles.tex}
\input{../../../references/assets/tikz/fonts-serif.tex}
\begin{document}
\begin{tikzpicture}
  % ---- 第一列：招募与分组（主轴向下，剔除支线在左）----
  \node[pc start]    (recruit) at (0,0)      {招募研究对象};
  \node[pc decision] (include) at (0,-2.3)   {符合纳入标准？};
  \node[pc process]  (excl)    at (-3.0,-2.3){剔除并记录原因};
  \node[pc key]      (random)  at (3.0,-2.3) {随机分组};
  % ---- 第二列：干预与随访（双臂上下分列，脱落支线在下）----
  \node[pc process] (ctrl)   at (6.2,-0.9) {对照组：常规方案};
  \node[pc process] (treat)  at (6.2,-3.7) {干预组：强化方案};
  \node[pc process] (follow) at (9.3,-2.3) {随访 24 周与数据采集};
  % ---- 第三列：统计与结论（正态性回路在右，下行为主）----
  \node[pc process]  (clean) at (12.8,-2.3) {数据汇总与清洗};
  \node[pc decision] (norm)  at (12.8,-4.6) {满足正态性？};
  \node[pc process]  (trans) at (16.0,-4.6) {数据变换};
  \node[pc key]      (test)  at (12.8,-7.0) {组间显著性检验};
  \node[pc process]  (effect)at (12.8,-9.1) {效应量与 95\%CI};
  \node[pc start]    (concl) at (12.8,-11.0){研究结论};

  % ---- 阶段容器 ----
  \node[pc sub, fit=(recruit)(include)(excl)(random), inner sep=9pt,
        label={[pc sub label]above:（一）招募与分组}] {};
  \node[pc sub, fit=(ctrl)(treat)(follow), inner sep=9pt,
        label={[pc sub label]above:（二）干预与随访}] {};
  \node[pc sub, fit=(clean)(norm)(trans)(test)(effect)(concl), inner sep=9pt,
        label={[pc sub label]above:（三）统计与结论}] {};

  % ---- 主流程：自西向东 ----
  \draw[pc flow] (recruit.south) -- (include.north);
  \draw[pc flow] (include.west) -- node[pc label, above=1pt]{否} (excl.east);
  \draw[pc flow] (include.east) -- node[pc label, above=1pt]{是} (random.west);
  % 随机分组：一线分双臂（北/南进入）
  \draw[pc flow] (random.east) -| (ctrl.south);
  \draw[pc flow] (random.east) -| (treat.north);
  % 双臂汇入随访：从北侧/南侧进入
  \draw[pc flow] (ctrl.east)  -| (follow.north);
  \draw[pc flow] (treat.east) -| (follow.south);
  % 随访 → 清洗 → 正态性
  \draw[pc flow] (follow.east) -- (clean.west);
  \draw[pc flow] (clean.south) -- (norm.north);
  % 正态性回路：否 → 变换 → 变换后重新检验（绕外侧回流）
  \draw[pc flow] (norm.east) -- node[pc label, above=1pt]{否} (trans.west);
  \draw[pc flow back] (trans.south) |- node[pc label, above=1pt, pos=0.75]{变换后重新检验} (test.east);
  \draw[pc flow] (norm.south) -- node[pc label, right=3pt]{是} (test.north);
  \draw[pc flow] (test) -- (effect);
  \draw[pc flow] (effect) -- (concl);
\end{tikzpicture}
\end{document}
