% ============================================================
% pretty-charts 字体层：演示版（PPT / T3）
% 用法：\input{.../references/style/tikz/fonts-ppt.tex}
%
% 与 fonts-paper.tex 同理，按字体是否存在自动回退：
%   Windows → Arial + Microsoft YaHei
%   其他平台 → TeX Gyre Heros + Noto Sans CJK SC
% 注意：Microsoft YaHei 是静态字体，可以安全嵌入 PDF；Noto Sans SC 在 Windows
% 上多为可变字体，xdvipdfmx 无法嵌入（见 spec/type.md §3），故这里不做回退。
% ============================================================
\IfFontExistsTF{Arial}
  {\setmainfont{Arial}}
  {\setmainfont{TeX Gyre Heros}}           % TeX Live 自带，Helvetica/Arial 的等价物
\IfFontExistsTF{Microsoft YaHei}
  {\setCJKmainfont{Microsoft YaHei}}       % 中文黑体（Windows，静态字体）
  {\setCJKmainfont{Noto Sans CJK SC}}      % Linux: apt install fonts-noto-cjk
